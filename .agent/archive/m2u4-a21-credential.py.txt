A21 red credential, preserved from `wt/rev-m2u4` before that branch was removed.

`.py.txt` keeps it out of pytest collection (`testpaths = ["tests"]`) and out of the
type checker while the polish item stays open. Restore it into `tests/` when the item
is taken up. It is RED against current `scripts/regenerate_r_clinical_goldens.py`: the
generator copies `_expected_outputs(stem)` out of a staging directory it then deletes,
so an unexpected producer file, temp file or symlink is never observed, and
`regenerate()` returns success instead of raising.

Acceptance check for the fix lives in `.agent/polish.md`.

from __future__ import annotations

import importlib.util
import pathlib

import pytest

_ROOT = pathlib.Path(__file__).resolve().parent.parent
_GENERATOR = _ROOT / "scripts" / "regenerate_r_clinical_goldens.py"


def _load_generator():
    spec = importlib.util.spec_from_file_location("r_clinical_golden_generator", _GENERATOR)
    assert spec is not None
    assert spec.loader is not None
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def test_golden_regeneration_rejects_unexpected_output(
    tmp_path: pathlib.Path, monkeypatch: pytest.MonkeyPatch
) -> None:
    generator = _load_generator()
    run_clinical = generator._run_clinical

    def emit_extra(source: pathlib.Path) -> None:
        run_clinical(source)
        source.with_name(f"{source.stem}_unexpected.tmp").write_text("unexpected\n")

    monkeypatch.setattr(generator, "_run_clinical", emit_extra)
    with pytest.raises(RuntimeError, match="unexpected"):
        generator.regenerate(tmp_path / "generated")
